+const hits = new Map<string, number>();
+
+export function rateLimit(key: string, limit: number, windowMs: number): boolean {
+  const now = Date.now();
+  const last = hits.get(key) ?? 0;
+  if (now - last < windowMs && (hits.get(key + ':count') ?? 0) >= limit) return false;
+  hits.set(key, now);
+  hits.set(key + ':count', ((hits.get(key + ':count') ?? 0) + 1));
+  return true;
+}
